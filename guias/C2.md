# 1. Liberar espacio en disco eliminando temporales y backups antiguos
rm -rf /tmp/* /tmp/.* /root/hst_install_backups/* /root/hst_backups/* 2>/dev/null
apt-get clean
apt-get autoremove -y
rm -rf /var/cache/apt/archives/*.deb

# 2. Instalar Node.js 24, npm, git y utilidades de compilación
curl -fsSL https://deb.nodesource.com/setup_24.x | bash -
apt-get update && apt-get install -y nodejs npm git build-essential

# 3. Clonar en /root y asignar permisos de ejecución (chmod +x)
rm -rf /root/hestiacp /root/hestiacp-src
git clone https://github.com/pirulug/hestiacp.git /root/hestiacp
chmod +x /root/hestiacp/src/*.sh
chmod +x /root/hestiacp/bin/*
chmod +x /root/hestiacp/install/*.sh

# 4. Definir directorio de compilación en /root y compilar
export BUILD_DIR=/root/hestiacp-src
cd /root/hestiacp/src
./hst_autocompile.sh --hestia --install '~localsrc'

# 5. Corregir permisos del Gestor de Archivos (File Manager)
/usr/local/hestia/bin/v-add-sys-filemanager
chown -R root:root /usr/local/hestia/web/fm
chown -R hestiaweb:hestiaweb /usr/local/hestia/web/fm/private
chown -R hestiaweb:hestiaweb /usr/local/hestia/web/fm/repository
chmod -R 775 /usr/local/hestia/web/fm/private
chmod -R 775 /usr/local/hestia/web/fm/repository

# 6. Reiniciar el servicio web del panel
systemctl restart hestia

--- Cargar VPN

# 1. Limpiar y clonar la version actualizada del repositorio
rm -rf /root/hestiacp
git clone https://github.com/pirulug/hestiacp.git /root/hestiacp

# 2. Copiar scripts ejecutables y librerias del backend
cp -f /root/hestiacp/bin/v-*vpn* /usr/local/hestia/bin/
cp -f /root/hestiacp/bin/v-list-sys-services /usr/local/hestia/bin/
cp -f /root/hestiacp/bin/v-start-service /usr/local/hestia/bin/
cp -f /root/hestiacp/bin/v-stop-service /usr/local/hestia/bin/
cp -f /root/hestiacp/bin/v-restart-service /usr/local/hestia/bin/
cp -f /root/hestiacp/bin/v-update-firewall /usr/local/hestia/bin/
cp -f /root/hestiacp/func/vpn.sh /usr/local/hestia/func/
cp -f /root/hestiacp/func/main.sh /usr/local/hestia/func/

# 3. Asignar permisos de ejecucion y propiedad a los binarios
chmod +x /usr/local/hestia/bin/v-*
chmod +x /usr/local/hestia/func/*.sh
chown -R root:root /usr/local/hestia/bin/
chown -R root:root /usr/local/hestia/func/

# 4. Copiar controladores y plantillas de la interfaz web
mkdir -p /usr/local/hestia/web/list/vpn /usr/local/hestia/web/add/vpn /usr/local/hestia/web/edit/vpn /usr/local/hestia/web/delete/vpn /usr/local/hestia/web/suspend/vpn /usr/local/hestia/web/unsuspend/vpn /usr/local/hestia/web/download/vpn

cp -rf /root/hestiacp/web/add/vpn/* /usr/local/hestia/web/add/vpn/
cp -rf /root/hestiacp/web/edit/vpn/* /usr/local/hestia/web/edit/vpn/
cp -rf /root/hestiacp/web/delete/vpn/* /usr/local/hestia/web/delete/vpn/
cp -rf /root/hestiacp/web/suspend/vpn/* /usr/local/hestia/web/suspend/vpn/
cp -rf /root/hestiacp/web/unsuspend/vpn/* /usr/local/hestia/web/unsuspend/vpn/
cp -rf /root/hestiacp/web/download/vpn/* /usr/local/hestia/web/download/vpn/

cp -f /root/hestiacp/web/templates/pages/list_vpn.php /usr/local/hestia/web/templates/pages/
cp -f /root/hestiacp/web/templates/pages/add_vpn.php /usr/local/hestia/web/templates/pages/

systemctl restart hestia
